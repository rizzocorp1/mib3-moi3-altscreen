import gdb
st = {'base': None, 'pacer':0, 'gate':0, 'notready':0, 'fed':0, 'empty':0, 'feeder':0, 'fbd':0, 'fbd_nz':0}
def rd(a,n):
    try: return int.from_bytes(bytes(gdb.selected_inferior().read_memory(a,n)),'little')
    except Exception: return None
def reg(n):
    try: return int(gdb.parse_and_eval('$'+n)) & 0xffffffffffffffff
    except Exception: return None

class Pacer(gdb.Breakpoint):
    def stop(self):
        st['pacer'] += 1
        if st['pacer']==1: gdb.write('[DIAG] PACER rodou\n')
        return False

    def stop(self):
        st['feeder'] += 1
        if st['feeder']==1: gdb.write('[DIAG] FEEDER rodou (thread viva)\n')
        return False

    def stop(self):
        st['gate'] += 1
        if st['gate'] <= 8:
            x21 = reg('x21')

            if S:
                W  = rd(S+0x30,8); pb = rd(S+0x3e8,8); sz = rd(S+0x3f0,8)
                E  = rd(W+0x30,8) if W else None

                r1 = rd(E+0x11,1) if E else None

                          % (st['gate'], S, W or 0, pb or 0, sz, E or 0, r0, r1, e18, est))

        return False
