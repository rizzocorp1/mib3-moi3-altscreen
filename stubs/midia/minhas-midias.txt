# minhas-midias.txt — perfil da mídia de mentira (Marco 15, 12/10/2026).
# SIMULAÇÃO do começo ao fim: nenhum valor daqui foi lido do carro, de um pendrive ou de um celular.
# Não há aparelho, não há arquivo de música e não sai som. As faixas são inventadas, e os nomes
# dizem "TESTE" para ninguém as tomar por músicas de verdade.
# Formato: o do carro de mentira (stubs/veiculo/README.md), com os nomes do gerado/esquema.txt
# desta pasta. O que cada seção faz e o que a HMI espera estão no README.md desta pasta.

# Os pedidos set… que o motor atende sozinho, pela regra do esquema: o idioma da base
# (setPreferredLanguage) e a mídia e o modo do navegador (setBrowseMedia, setBrowseMode). Os do
# tocador e os outros do navegador quem atende é o modelo (driver/midiastub.c).
grava = DSIMediaBase DSIMediaBrowser

# --- A fonte --------------------------------------------------------------------------------------
# SIMULAÇÃO. Um aparelho do tipo 0 (DEVICETYPE_JUKEBOX), que a tela chama de "Minhas mídias". É um
# dos cinco tipos que esta HMI aceita como fonte de mídia (0, 4, 8, 10 e 15: DeviceFilter.MEDIA);
# pendrive (tipo 2) e cartão (tipo 1) ela não põe na lista "Fontes". Sem aparelho nenhum a HMI nem
# liga as fontes, e a tela fica no aviso "função indisponível".
[DSIMediaBase.MBLupdateDeviceList]

deviceList[0].deviceType = 0
deviceList[0].numSlots = 1
