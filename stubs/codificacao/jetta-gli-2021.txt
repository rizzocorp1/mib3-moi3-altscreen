# --- Adaptações do 5F, LIDAS DO CARRO em 14/10/2026 ---------------------------------------------

# Navegação: ligada, região MSA (América do Sul / Mercosul).
0558 @1.7+1 = 1      # ID_NAVIGATION_SYSTEM = active ("Navigation_System [VN]_active")
0558 @0.0+8 = 3      # ID_COUNTRY_NAVIGATION = MSA (0x03) ("Country_Navigation [VO]_MSA")

# Mapa no cluster por IP: é o coding que liga o stream do mapa para o painel (a nossa cluster-map).
0B5A @0.0+8 = 1      # ID_NAVIGATION_MAP_TRANSMISSION_MODE = IP Streaming (0x01)

# Rádio: faixa de FM do Brasil (confirma a dedução anterior).
0559 @1.4+4 = 6      # ID_BAND_SETTINGS_FM_TUNER = Brasil ("band_settings_fm_tuner [VO]_BRA")
# AM no autoscan = NAR; o numérico do campo AM (0559 @2.4+4) não foi mapeado aqui — a HMI usa o
# padrão. Conferir o código de AM se a tela do rádio AM for montada.

# App-Connect: CarPlay e Android Auto, os DOIS ligados no carro.
0555 @1.2+2 = 1      # ID_APPLE_DIO: CarPlay on ("apple_car_play [VN]_on"). A HMI só olha se é != 0;
                     # o valor cru exato (1 ou 2) o autoscan não dá no modo por nome.
0555 @1.4+2 = 1      # ID_GOOGLE_ANDROID_AUTO: on ("google_android_auto [VN]_on"). CORRIGE a dedução
                     # anterior, que o deixava em 0 porque a foto não o mostrava.

# País de mídia (Verkaufsland): Brasil.

                     # = BR. Encoding como ASCII de 2 letras (convenção); o cru não foi reconferido.
